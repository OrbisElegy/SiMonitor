cmake_minimum_required(VERSION 3.20)
project(sim_audio_native VERSION 1.0 LANGUAGES C)
option(SIM_AUDIO_TEST "Build explicitly non-hardware test adapter" OFF)
set(MINIAUDIO_HEADER "${CMAKE_CURRENT_SOURCE_DIR}/vendor/miniaudio.h")
if(NOT EXISTS "${MINIAUDIO_HEADER}")
  message(FATAL_ERROR "Missing miniaudio: run python tools/fetch_dependencies.py from the repository root")
endif()
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/../../eng/dependencies.json" DEPENDENCIES)
string(JSON NATIVE_COUNT LENGTH "${DEPENDENCIES}" native_dependencies)
math(EXPR NATIVE_LAST "${NATIVE_COUNT} - 1")
foreach(INDEX RANGE ${NATIVE_LAST})
  string(JSON DEPENDENCY_ID GET "${DEPENDENCIES}" native_dependencies ${INDEX} id)
  if(DEPENDENCY_ID STREQUAL "miniaudio")
    string(JSON EXPECTED_HASH GET "${DEPENDENCIES}" native_dependencies ${INDEX} source_files 0 sha256)
  endif()
endforeach()
file(SHA256 "${MINIAUDIO_HEADER}" ACTUAL_HASH)
if(NOT ACTUAL_HASH STREQUAL EXPECTED_HASH)
  message(FATAL_ERROR "miniaudio hash mismatch: run python tools/fetch_dependencies.py")
endif()
add_library(sim_audio_native SHARED sim_audio.c)
set_target_properties(sim_audio_native PROPERTIES C_STANDARD 99 C_VISIBILITY_PRESET hidden)
if(SIM_AUDIO_TEST)
  target_compile_definitions(sim_audio_native PRIVATE SIM_AUDIO_TEST)
endif()
if(NOT WIN32)
  target_link_libraries(sim_audio_native PRIVATE m pthread dl)
endif()
if(SIM_AUDIO_TEST)
  include(CTest)
  if(BUILD_TESTING)
    add_executable(sim_audio_retirement_tests tests/check_retirement.c)
    set_target_properties(sim_audio_retirement_tests PROPERTIES C_STANDARD 99 C_VISIBILITY_PRESET hidden)
    target_compile_definitions(sim_audio_retirement_tests PRIVATE SIM_AUDIO_TEST)
    if(NOT WIN32)
      target_link_libraries(sim_audio_retirement_tests PRIVATE m pthread dl)
    endif()
    add_test(NAME sim_audio_retirement COMMAND sim_audio_retirement_tests)
  endif()
endif()
